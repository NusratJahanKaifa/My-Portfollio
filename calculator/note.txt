Making Calculator:
<!DOCTYPE html>
<head>
<body class=" flex justify-center items-start min-h-screen pt-10">
<div class=" p-6 rounded-lg shadow-lg w-80 text-center">
<h1 class="text-3xl font-bold "> Calculator </h1>
<h2  id="result" class=" font-bold "> 0 </h2>            
<input type="number" id="num1">
<input type="number" id="num2">
<button onclick= "Calculate('Sum')" > Summation </button>
<button onclick= "Calculate('Subtract')" > Subtraction </button>
<button onclick= "Calculate('Divide')" > Divition </button>
<button onclick= "Calculate('Multiply')" > Multipication </button>
</div>
<script>
    function Calculate(type){
    let num1 = parseInt(document.getElementById("num1").value); // parse int ar kaj hocce input box a je data likbo tha string data setake int a anar jonno parseint use krte hoi
    let num2 = parseInt(document.getElementById("num2").value);  // initialy value 0 dekar jonno 
    let result = document.getElementById("result");
if(type == 'Sum'){
    result.innerHTML = (num1 + num2)}
else if ( type == 'Subtract'){
    result.innerHTML = (num1 - num2)}
else if ( type == 'Divide'){
    result.innerHTML = (num1 / num2)}
else{
    result.innerHTML = (num1 * num2)
}
}
</script>
</body>
